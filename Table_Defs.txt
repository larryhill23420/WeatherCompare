CREATE TABLE ZipCode (
    ZipCode        VARCHAR(10)  NOT NULL,
    PlaceName      VARCHAR(100) NOT NULL,
    Latitude       DECIMAL(9,6),
    Longitude      DECIMAL(9,6),
    TimeZone       VARCHAR(50),          -- e.g. 'America/Los_Angeles'
    UtcOffsetSeconds INT,
    InsertedAt     VARCHAR(25),
    PRIMARY KEY (ZipCode, PlaceName)
);

CREATE TABLE Reading (
    Zip_DateTime   VARCHAR(50)  NOT NULL,  -- e.g. '98593_2026-11-21T08:00:00Z'
    DateTime       VARCHAR(25)  NOT NULL,  -- UTC timestamp
    ZipCode        VARCHAR(10),
    PlaceName      VARCHAR(100),
    Temperature    INT,                    -- °F
    Precipitation  DECIMAL(5,2),
    Humidity       INT,
    CloudCover     INT,
    InsertedAt     VARCHAR(25),
    PRIMARY KEY (Zip_DateTime, DateTime)
);

CREATE TABLE Error (
    UUID           VARCHAR(36)  NOT NULL,
    Zip_DateTime   VARCHAR(50)  NOT NULL,
    Description    VARCHAR(1000),
    InsertedAt     VARCHAR(25),
    PRIMARY KEY (UUID, Zip_DateTime)
);
